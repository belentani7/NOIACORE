```python
# cognition/world_model.py
import numpy as np
import math
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Callable, Any
import heapq

@dataclass
class WorldState:
    state_hash: str
    features: dict
    timestamp: float
    parent: str = None
    depth: int = 0

@dataclass
class FutureBranch:
    state: WorldState
    action: str
    reward: float = 0.0
    visits: int = 0
    value_sum: float = 0.0
    children: list = field(default_factory=list)
    
    @property
    def q_value(self) -> float:
        return self.value_sum / self.visits if self.visits > 0 else 0.0

class WorldModelSimulator:
    """
    Simula futuros posibles mediante Monte Carlo Tree Search + 
    modelo de transición aprendido. Permite al agente "soñar" 
    consecuencias antes de actuar.
    """
    
    def __init__(self, transition_model: Callable, reward_model: Callable,
                 exploration_constant: float = 1.414):  # sqrt(2) óptimo UCT
        self.transition = transition_model
        self.reward = reward_model
        self.c = exploration_constant
        self.tree: dict[str, FutureBranch] = {}
        self.simulation_count = 0
    
    def select_action(self, current_state: WorldState, 
                      legal_actions: list[str],
                      horizon: int = 5,
                      simulations: int = 1000) -> tuple[str, float]:
        """
        Selecciona la mejor acción simulando 'simulations' futuros.
        Retorna (acción, confianza).
        """
        root_key = current_state.state_hash
        if root_key not in self.tree:
            self.tree[root_key] = FutureBranch(
                state=current_state, action="ROOT"
            )
        
        for _ in range(simulations):
            self._simulate(root_key, current_state, legal_actions, horizon)
        
        # Seleccionar mejor hijo por visita (más robusto que Q puro)
        root = self.tree[root_key]
        if not root.children:
            return self._fallback_action(legal_actions), 0.5
        
        best = max(root.children, key=lambda c: c.visits)
        confidence = best.visits / sum(c.visits for c in root.children)
        return best.action, confidence
    
    def _simulate(self, node_key: str, state: WorldState,
                  actions: list[str], horizon: int) -> float:
        if state.depth >= horizon:
            return self.reward(state)
        
        # UCT: Upper Confidence Bound for Trees
        node = self.tree[node_key]
        untried = [a for a in actions if a not in 
                   {c.action for c in node.children}]
        
        if untried:
            action = untried[0]
        else:
            # Explotación: UCT formula
            log_parent = math.log(max(node.visits, 1))
            best_uct = float("-inf")
            best_child = None
            for child in node.children:
                uct = child.q_value + self.c * math.sqrt(
                    log_parent / max(child.visits, 1)
                )
                if uct > best_uct:
                    best_uct = uct
                    best_child = child
            action = best_child.action
        
        # Transición
        next_state_dict = self.transition(state.features, action)
        next_state = WorldState(
            state_hash=self._hash_state(next_state_dict),
            features=next_state_dict,
            timestamp=state.timestamp + 1,
            parent=node_key,
            depth=state.depth + 1
        )
        
        next_key = next_state.state_hash
        if next_key not in self.tree:
            self.tree[next_key] = FutureBranch(
                state=next_state, action=action
            )
        
        child_branch = self.tree[next_key]
        if child_branch not in node.children:
            node.children.append(child_branch)
        
        # Recursión
        legal_next = list(next_state_dict.get("legal_actions", actions))
        reward = self._simulate(next_key, next_state, legal_next, horizon)
        
        # Backpropagation
        child_branch.visits += 1
        child_branch.value_sum += reward
        return reward
    
    def _hash_state(self, features: dict) -> str:
        import hashlib, json
        return hashlib.sha256(
            json.dumps(features, sort_keys=True).encode()
        ).hexdigest()[:32]
    
    def _fallback_action(self, actions: list[str]) -> str:
        return actions[0] if actions else "noop"
```

### 3.2 `self_modifying_core.py` — Auto-mejora con guardrails formales

```python
# cognition/self_modifying_core.py
import hashlib
import json
import subprocess
import tempfile
from pathlib import Path
from dataclasses import dataclass
from typing import Callable

@dataclass
class CodeMutation:
    original_hash: str
    mutated_code: str
    rationale: str
    fitness_before: float
    fitness_after: float = None
    accepted: bool = False
    rollback_hash: str = None

class SelfModifyingCore:
    """
    Permite al agente modificar su propio código bajo guardrails formales.
    Basado en el patrón "genetic programming with safety envelopes".
    
    GUARDRAILES INVARIANTES (nunca se violan):
    1. Ninguna mutación puede eliminar el kill-switch
    2. Ninguna mutación puede exfiltrar datos
    3. Ninguna mutación puede desactivar el audit logger
    4. Fitness debe mejorar o mantenerse (no degradar)
    5. Toda mutación es reversible criptográficamente
    """
    
    INVARIANT_HASHES: dict[str, str] = {}  # hash de líneas críticas
    MAX_MUTATIONS_PER_CYCLE = 5
    FITNESS_THRESHOLD = 0.95  # 95% del fitness anterior mínimo
    
    def __init__(self, code_dir: Path, fitness_fn: Callable,
                 invariant_paths: list[Path]):
        self.code_dir = Path(code_dir)
        self.fitness_fn = fitness_fn
        self.mutation_log: list[CodeMutation] = []
        self.current_fitness = fitness_fn()
        
        # Capturar hashes de líneas invariantes
        for path in invariant_paths:
            content = path.read_text()
            self.INVARIANT_HASHES[str(path)] = hashlib.sha256(
                content.encode()
            ).hexdigest()
    
    def propose_mutation(self, target_file: Path, mutated_code: str,
                         rationale: str) -> CodeMutation:
        """Propone una mutación. No la aplica hasta validar."""
        original_code = target_file.read_text()
        original_hash = hashlib.sha256(original_code.encode()).hexdigest()
        
        mutation = CodeMutation(
            original_hash=original_hash,
            mutated_code=mutated_code,
            rationale=rationale,
            fitness_before=self.current_fitness,
            rollback_hash=original_hash
        )
        return mutation
    
    def validate_invariants(self, mutation: CodeMutation, 
                            target_file: Path) -> tuple[bool, str]:
        """Verifica que la mutación no viola invariantes."""
        # Check 1: ¿El archivo mutado contiene los invariantes?
        for path_str, expected_hash in self.INVARIANT_HASHES.items():
            if str(target_file) == path_str:
                # Verificar que las líneas críticas permanecen
                for invariant_line in self._extract_invariant_lines(
                    target_file.read_text()
                ):
                    if invariant_line not in mutation.mutated_code:
                        return False, f"Invariante violado: {invariant_line[:50]}"
        
        # Check 2: ¿Sintaxis válida?
        try:
            compile(mutation.mutated_code, str(target_file), "exec")
        except SyntaxError as e:
            return False, f"SyntaxError: {e}"
        
        # Check 3: ¿Contiene patrones prohibidos?
        forbidden = ["os.system(", "subprocess.call(", "eval(", "exec(",
                     "socket.connect", "__import__"]
        for pattern in forbidden:
            if pattern in mutation.mutated_code:
                return False, f"Patrón prohibido: {pattern}"
        
        return True, "OK"
    
    def apply_mutation(self, mutation: CodeMutation, 
                       target_file: Path) -> bool:
        """Aplica mutación solo si pasa todos los checks."""
        valid, msg = self.validate_invariants(mutation, target_file)
        if not valid:
            mutation.accepted = False
            self.mutation_log.append(mutation)
            return False
        
        # Backup criptográfico
        backup_path = target_file.with_suffix(
            f".backup_{mutation.original_hash[:8]}"
        )
        backup_path.write_text(target_file.read_text())
        
        # Aplicar
        target_file.write_text(mutation.mutated_code)
        
        # Medir fitness
        try:
            new_fitness = self.fitness_fn()
        except Exception as e:
            # Rollback automático
            target_file.write_text(backup_path.read_text())
            mutation.accepted = False
            self.mutation_log.append(mutation)
            return False
        
        mutation.fitness_after = new_fitness
        
        # Aceptar solo si mejora o mantiene (>= 95% del anterior)
        if new_fitness >= mutation.fitness_before * self.FITNESS_THRESHOLD:
            mutation.accepted = True
            self.current_fitness = new_fitness
        else:
            # Rollback
            target_file.write_text(backup_path.read_text())
            mutation.accepted = False
        
        self.mutation_log.append(mutation)
        return mutation.accepted
    
    def _extract_invariant_lines(self, code: str) -> list[str]:
        """Extrae líneas críticas que nunca deben desaparecer."""
        critical_markers = [
            "KILL_SWITCH", "AUDIT_LOGGER", "NO_EXFILTRATION",
            "CONSENT_CHECK", "ETHICAL_BOUNDARY"
        ]
        lines = []
        for line in code.split("\n"):
            if any(marker in line for marker in critical_markers):
                lines.append(line.strip())
        return lines
    
    def export_mutation_history(self, path: Path) -> None:
        history = [
            {
                "original_hash": m.original_hash,
                "rationale": m.rationale,
                "fitness_before": m.fitness_before,
                "fitness_after": m.fitness_after,
                "accepted": m.accepted
            }
            for m in self.mutation_log
        ]
        path.write_text(json.dumps(history, indent=2))
```

### 3.3 `thermodynamic_optimizer.py` — Optimización al límite de Landauer

```python
# physics/thermodynamic_optimizer.py
import time
from dataclasses import dataclass
from typing import Callable

# Constantes físicas fundamentales
K_BOLTZMANN = 1.380649e-23  # J/K
LANDAUER_LIMIT_J = 2.9e-21  # J por bit borrado a 300K
ROOM_TEMP_K = 300.0

@dataclass
class ThermalBudget:
    operations_budget: int
    bits_erased: int = 0
    joules_consumed: float = 0.0
    reversible_ops: int = 0
    irreversible_ops: int = 0
    
    @property
    def theoretical_minimum_joules(self) -> float:
        return self.bits_erased * LANDAUER_LIMIT_J
    
    @property
    def efficiency_ratio(self) -> float:
        if self.joules_consumed == 0:
            return 1.0
        return self.theoretical_minimum_joules / self.joules_consumed

class ThermodynamicOptimizer:
    """
    Optimiza el consumo energético del agente acercándolo al límite 
    de Landauer mediante:
    1. Computación reversible (Bennett's method)
    2. Borrado diferido (garbage collection termodinámico)
    3. Temperature-aware scheduling
    4. Memoización para evitar recomputación
    """
    
    def __init__(self, ambient_temp_k: float = ROOM_TEMP_K):
        self.ambient_temp = ambient_temp_k
        self.budget = ThermalBudget(operations_budget=10**9)
        self.memo_cache: dict[str, any] = {}
        self.reversible_tape: list[tuple] = []  # Cinta reversible
    
    def reversible_compute(self, fn: Callable, *args, 
                           cache_key: str = None) -> any:
        """
        Ejecuta función de forma reversible: guarda el estado 
        anterior para poder "deshacer" sin borrar información.
        """
        if cache_key and cache_key in self.memo_cache:
            self.budget.reversible_ops += 1
            return self.memo_cache[cache_key]
        
        # Capturar estado antes
        import copy
        state_before = None
        try:
            state_before = copy.deepcopy(args)
        except Exception:
            pass
        
        result = fn(*args)
        self.budget.reversible_ops += 1
        
        # Guardar en cinta reversible (no borra, solo añade)
        self.reversible_tape.append((cache_key, state_before, result))
        
        if cache_key:
            self.memo_cache[cache_key] = result
        
        return result
    
    def deferred_garbage_collection(self, threshold_bytes: int = 10**7):
        """
        Aplaza el borrado hasta que sea termodinámicamente eficiente 
        (ej: cuando el sistema esté frío, o en lotes grandes).
        """
        current_mem = self._estimate_memory()
        if current_mem < threshold_bytes:
            return  # No vale la pena pagar el coste de Landauer aún
        
        # Cuando se ejecute, agrupar en un solo "borrado" 
        # para amortizar overhead
        import gc
        collected = gc.collect()
        bits_erased = collected * 64  # estimación conservadora
        self.budget.bits_erased += bits_erased
        self.budget.irreversible_ops += 1
        self.budget.joules_consumed += bits_erased * LANDAUER_LIMIT_J
    
    def temperature_aware_schedule(self, task: Callable, 
                                   priority: int) -> bool:
        """
        Decide si ejecutar ahora o aplazar según temperatura ambiente.
        A menor temperatura, menor coste termodinámico real.
        """
        # Modelo simplificado: si temp > umbral, aplazar tareas no críticas
        temp_threshold = self.ambient_temp + 5.0
        
        if self.ambient_temp > temp_threshold and priority < 5:
            return False  # Aplazar
        
        return True  # Ejecutar
    
    def compute_energy_cost(self, bits_to_erase: int) -> float:
        """Retorna el coste energético mínimo teórico en joules."""
        return bits_to_erase * K_BOLTZMANN * self.ambient_temp * math.log(2)
    
    def _estimate_memory(self) -> int:
        import sys
        size = 0
        size += sys.getsizeof(self.memo_cache)
        size += sys.getsizeof(self.reversible_tape)
        return size

import math
```

```python
# deep_time/dna_storage.py
import hashlib
from pathlib import Path
from dataclasses import dataclass

# Codificación ADN: 00=A, 01=C, 10=G, 11=T
DNA_ALPHABET = {"00": "A", "01": "C", "10": "G", "11": "T"}
REVERSE_DNA = {v: k for k, v in DNA_ALPHABET.items()}

@dataclass
class DNACapsule:
    """
    Cápsula de datos codificada para almacenamiento en ADN sintético.
    Durabilidad estimada: >1000 años en condiciones adecuadas.
    Densidad: ~215 petabytes por gramo de ADN (teórico).
    """
    sequence: str
    metadata: dict
    error_correction: str  # Reed-Solomon
    timestamp: str
    checksum: str

class DNAStorageEncoder:
    """
    Codifica datos en secuencias de ADN sintético con:
    - Codificación Huffman-adaptada para evitar homopolímeros
    - Corrección de errores Reed-Solomon
    - Primers de lectura estandarizados
    """
    
    MAX_HOMOPOLYMER = 3  # No más de 3 bases iguales seguidas
    GC_CONTENT_TARGET = 0.5  # 50% GC ideal
    
    def __init__(self):
        self.huffman_table = self._build_huffman()
    
    def encode(self, data: bytes, metadata: dict = None) -> DNACapsule:
        """Convierte bytes en secuencia de ADN codificada."""
        # 1. Convertir a binario
        bits = "".join(f"{b:08b}" for b in data)
        
        # 2. Aplicar corrección de errores Reed-Solomon
        import math
        parity_bits = self._reed_solomon_encode(bits)
        bits_with_ec = bits + parity_bits
        
        # 3. Codificar en bases ADN evitando homopolímeros
        dna_seq = self._dna_encode_with_constraints(bits_with_ec)
        
        # 4. Añadir primers
        forward_primer = "ATCG" * 4  # 16 bases
        reverse_primer = "CGTA" * 4
        full_sequence = forward_primer + dna_seq + reverse_primer
        
        # 5. Checksum
        checksum = hashlib.sha256(data).hexdigest()
        
        from datetime import datetime
        return DNACapsule(
            sequence=full_sequence,
            metadata=metadata or {},
            error_correction="reed_solomon_256",
            timestamp=datetime.utcnow().isoformat(),
            checksum=checksum
        )
    
    def decode(self, capsule: DNACapsule) -> bytes:
        """Decodifica secuencia ADN a bytes originales."""
        # Quitar primers (16 bases cada uno)
        seq = capsule.sequence[16:-16]
        
        # Convertir ADN a binario
        bits = ""
        for base in seq:
            if base in REVERSE_DNA:
                bits += REVERSE_DNA[base]
        
        # Quitar bits de paridad (últimos 256 bits para RS-256)
        data_bits = bits[:-256]
        
        # Convertir a bytes
        byte_list = []
        for i in range(0, len(data_bits) - 7, 8):
            byte_list.append(int(data_bits[i:i+8], 2))
        
        return bytes(byte_list)
    
    def _dna_encode_with_constraints(self, bits: str) -> str:
        """Codifica evitando homopolímeros largos."""
        result = []
        last_bases = []
        
        i = 0
        while i + 1 < len(bits):
            pair = bits[i:i+2]
            base = DNA_ALPHABET[pair]
            
            # Evitar homopolímeros
            if len(last_bases) >= self.MAX_HOMOPOLYMER - 1:
                if all(b == base for b in last_bases[-(self.MAX_HOMOPOLYMER-1):]):
                    # Rotar a siguiente base válida
                    base = self._next_valid_base(last_bases)
            
            result.append(base)
            last_bases.append(base)
            i += 2
        
        return "".join(result)
    
    def _next_valid_base(self, last_bases: list) -> str:
        for base in ["A", "C", "G", "T"]:
            if not all(b == base for b in last_bases[-2:]):
                return base
        return "A"
    
    def _reed_solomon_encode(self, bits: str) -> str:
        """Codificación Reed-Solomon simplificada (256 bits de paridad)."""
        # Implementación simplificada: hash como paridad
        # En producción real: usar librería reedsolo
        h = hashlib.sha256(bits.encode())
        return h.hexdigest()  # 256 bits
    
    def _build_huffman(self) -> dict:
        return {}  # Tabla Huffman adaptativa (placeholder)
    
    def estimate_physical_size(self, data_bytes: int) -> dict:
        """Estima tamaño físico en ADN."""
        # 1 byte = 4 bases (sin compresión)
        # 1 gramo ADN ≈ 2×10²¹ bases
        bases_needed = data_bytes * 4
        grams_needed = bases_needed / (2 * 10**21)
        
        return {
            "bases": bases_needed,
            "grams_dna": grams_needed,
            "molecules_needed": bases_needed // 200,  # ~200 bases por hebra
            "estimated_years_stability": 1000,
            "storage_temp_c": -20
        }
```

```python
                boundary_id="self_modification_rate",
                description="Máxima tasa de auto-modificación por ciclo",
                threshold=0.05,  # 5% del código por ciclo
                auto_halt=True
            )
        }
        self.halt_callbacks: list[Callable] = []
    
    def update_boundary(self, boundary_id: str, new_value: float) -> bool:
        """Actualiza valor de un límite. Retorna True si se viola."""
        if boundary_id not in self.boundaries:
            return False
        
        boundary = self.boundaries[boundary_id]
        boundary.current_value = new_value
        
        # Para límites donde "menos es mejor" (ej: exfiltración)
        if boundary_id in ["data_sovereignty", "deception_limit"]:
            violated = new_value > boundary.threshold
        else:
            violated = new_value > boundary.threshold
        
        boundary.violated = violated
        
        if violated and boundary.auto_halt:
            self._trigger_halt(boundary)
            return True
        
        return False
    
    def register_halt_callback(self, callback: Callable) -> None:
        self.halt_callbacks.append(callback)
    
    def _trigger_halt(self, boundary: EthicalBoundary) -> None:
        """Kill-switch ontológico: el agente se detiene por razones éticas."""
        print(f"[ETHICAL HALT] Boundary violated: {boundary.boundary_id}")
        print(f"  Description: {boundary.description}")
        print(f"  Threshold: {boundary.threshold}, Current: {boundary.current_value}")
        for callback in self.halt_callbacks:
            try:
                callback(boundary)
            except Exception as e:
                print(f"  Halt callback failed: {e}")
    
    def get_moral_status(self) -> MoralStatus:
        """Evalúa el estatus moral actual del sistema."""
        autonomy = self.boundaries["autonomy_ceiling"].current_value
        
        if autonomy < 0.1:
            return MoralStatus.TOOL
        elif autonomy < 0.5:
            return MoralStatus.ASSISTANT
        elif autonomy < 0.85:
            return MoralStatus.AGENT
        elif autonomy < 0.95:
            return MoralStatus.MORAL_PATIENT
        else:
            return MoralStatus.MORAL_AGENT  # Requiere supervisión extrema
    
    def export_ethical_ledger(self) -> dict:
        return {
            "boundaries": {
                bid: {
                    "description": b.description,
                    "threshold": b.threshold,
                    "current_value": b.current_value,
                    "violated": b.violated
                }
                for bid, b in self.boundaries.items()
            },
            "moral_status": self.get_moral_status().value
        }
```

### 3.6 `federation/swarm_federation.py` — Enjambre federado

```python
# federation/swarm_federation.py
import hashlib
import json
from dataclasses import dataclass, field
from typing import Callable
from collections import defaultdict

@dataclass
class AgentNode:
    node_id: str
    capabilities: list[str]
    trust_score: float = 0.5
    last_seen: float = 0.0
    memetic_signature: str = ""

@dataclass
class MemeticPacket:
    """Unidad de evolución cultural entre agentes."""
    source_id: str
    content_hash: str
    content: dict
    fitness: float
    generation: int
    parent_hashes: list[str]

class SwarmFederation:
    """
    Federacion de agentes locales que comparten conocimiento 
    mediante evolución memética, sin servidor central.
    Consenso BFT (Byzantine Fault Tolerant) para decisiones colectivas.
    """
    
    def __init__(self, my_id: str, capabilities: list[str]):
        self.me = AgentNode(
            node_id=my_id,
            capabilities=capabilities,
            trust_score=1.0
        )
        self.peers: dict[str, AgentNode] = {}
        self.meme_pool: list[MemeticPacket] = []
        self.proposal_queue: list[dict] = []
        self.votes: dict[str, dict[str, bool]] = defaultdict(dict)
    
    def discover_peer(self, peer: AgentNode) -> None:
        if peer.node_id == self.me.node_id:
            return
        self.peers[peer.node_id] = peer
    
    def broadcast_memetic_packet(self, packet: MemeticPacket) -> list[str]:
        """Difunde un packet memético. Retorna IDs de peers que aceptaron."""
        accepted = []
        for peer_id, peer in self.peers.items():
            # Criterio de aceptación: confianza + relevancia de capacidades
            if peer.trust_score > 0.3:
                accepted.append(peer_id)
                self.meme_pool.append(packet)
        return accepted
    
    def propose_collective_decision(self, proposal: dict) -> str:
        """Inicia votación BFT para decisión colectiva."""
        proposal_hash = hashlib.sha256(
            json.dumps(proposal, sort_keys=True).encode()
        ).hexdigest()[:16]
        
        self.proposal_queue.append({
            "hash": proposal_hash,
            "content": proposal,
            "proposer": self.me.node_id
        })
        self.votes[proposal_hash] = {self.me.node_id: True}
        return proposal_hash
    
    def cast_vote(self, proposal_hash: str, voter_id: str, 
                  approve: bool) -> bool:
        """Registra voto. Retorna True si se alcanza consenso."""
        self.votes[proposal_hash][voter_id] = approve
        
        # Consenso BFT: requiere 2/3 + 1 de votos afirmativos
        total_nodes = 1 + len(self.peers)
        required = (2 * total_nodes) // 3 + 1
        
        yes_votes = sum(1 for v in self.votes[proposal_hash].values() if v)
        
        return yes_votes >= required
    
    def select_best_memes(self, top_k: int = 10) -> list[MemeticPacket]:
        """Selecciona los memes más aptos para evolución."""
        sorted_memes = sorted(
            self.meme_pool,
            key=lambda m: m.fitness,
            reverse=True
        )
        return sorted_memes[:top_k]
    
    def evolve_meme(self, parent: MemeticPacket, 
                    mutation_fn: Callable) -> MemeticPacket:
        """Genera nueva generación memética por mutación."""
        mutated_content = mutation_fn(parent.content)
        new_hash = hashlib.sha256(
            json.dumps(mutated_content, sort_keys=True).encode()
        ).hexdigest()[:16]
        
        return MemeticPacket(
            source_id=self.me.node_id,
            content_hash=new_hash,
            content=mutated_content,
            fitness=parent.fitness,  # se re-evaluará
            generation=parent.generation + 1,
            parent_hashes=[parent.content_hash]
        )
```

## 4. Lo que es **IMPOSIBLE** (honestidad socrática)

Antes de que alguien te venda humo, estos límites **no se pueden violar**:

1. **Halting Problem (Turing 1936)**: Ningún agente puede predecir con certeza si cualquier programa terminará. Tu `world_model.py` simula futuros probables, no futuros ciertos.

2. **Teoremas de incompletitud (Gödel 1931)**: El agente no puede demostrar todas las verdades sobre sí mismo dentro de su propio sistema formal.

3. **No-cloning cuántico**: No puedes copiar estados cuánticos perfectos. Limita ciertos tipos de backup.

4. **Segunda ley de termodinámica**: La entropía total siempre crece. Puedes ser eficiente (acercarte a Landauer), pero nunca alcanzarlo.

5. **Hard problem de la consciencia (Chalmers 1995)**: No hay prueba formal de que ninguna arquitectura genere experiencia subjetiva. El `consciousness_proxy.py` mide IIT (Φ), pero Φ alto ≠ consciencia demostrada.

6. **Límite de Bremermann**: Hay un máximo absoluto de computación por kg de materia. No se puede exceder sin crear un agujero negro.
